%  Es el "zoom" a la parte azul (E1) de la Figura 1: qué hay por dentro del
%  generador de imagen y del transmisor, a nivel de registros, contadores,
%  decodificadores y multiplexores.
%  Preámbulo necesario: \usetikzlibrary{positioning,arrows.meta,calc,fit,backgrounds}

\begin{tikzpicture}[
    font=\footnotesize,
    >={Stealth[length=2mm]},
    reg/.style={draw, fill=blue!10, minimum width=2.3cm, minimum height=0.9cm, align=center, line width=0.6pt},
    cmb/.style={draw, fill=white, minimum width=2.0cm, minimum height=0.8cm, align=center, line width=0.6pt},
    dec/.style={draw, fill=yellow!18, minimum width=1.5cm, minimum height=0.85cm, align=center, line width=0.6pt},
    fsm/.style={draw, fill=red!10, rounded corners=6pt, minimum width=2.9cm, minimum height=1.2cm, align=center, line width=0.6pt},
    mux/.style={draw, fill=green!12, minimum width=0.9cm, align=center, inner sep=2pt, line width=0.6pt},
    ext/.style={draw, fill=gray!8, align=center, minimum height=0.9cm, inner sep=3pt},
    sig/.style={->, line width=0.6pt},
    bus/.style={->, line width=1.4pt},
    lbl/.style={font=\scriptsize, align=center, inner sep=1.5pt},
    caja/.style={draw, dashed, rounded corners, inner sep=8pt},
    tit/.style={font=\small\bfseries},
]

% ======================= (A) Posición del punto =======================
\node[ext] (clk) at (0.2, 9.0) {CLK\_SYS\\50 MHz};
\node[cmb] (div) at (3.0, 9.0) {Divisor\\(generador\\de ticks)};
\node[reg] (pos) at (6.6, 9.0) {Contador de\\posición\\8 bits (pos)};
\draw[sig] (clk) -- (div);
\draw[sig] (div) -- node[lbl, above] {tick\_paso} (pos);
\node[lbl] at ($(div.east)!0.5!(pos.west)+(0,-0.3)$) {cada 60 ms};
\draw[sig] ([xshift=6mm]pos.north) -- ++(0,0.45) -| node[lbl, pos=0.25, above] {pos + 1} ([xshift=-6mm]pos.north);

% ======================= (B) Generador de imagen =======================
\node[dec] (dy) at (10.7, 10.3) {Deco\\3 a 8};
\node[dec] (dc) at (10.7, 9.0)  {Deco\\2 a 4};
\node[dec] (db) at (10.7, 7.7)  {Deco\\3 a 8};
\draw[bus] ([yshift=2.5mm]pos.east) -- ++(1.0,0) |- node[lbl, pos=0.68, above] {Y[2:0]} (dy.west);
\draw[bus] (pos.east) -- node[lbl, pos=0.72, above] {X[4:3]} (dc.west);
\draw[bus] ([yshift=-2.5mm]pos.east) -- ++(1.0,0) |- node[lbl, pos=0.68, below] {X[2:0]} (db.west);

\node[cmb, minimum height=3.4cm, minimum width=2.4cm] (and) at (13.7, 9.0)
     {Matriz de\\256 compuertas\\AND de 3\\entradas\\[4pt]$\mathrm{LED}=C_m\, F_d\, B_b$};
\draw[bus] (dy.east) -- node[lbl, above] {F[7:0]} (dy.east -| and.west);
\draw[bus] (dc.east) -- node[lbl, above] {C[3:0]} (dc.east -| and.west);
\draw[bus] (db.east) -- node[lbl, above] {B[7:0]} (db.east -| and.west);

\node[mux, minimum height=1.6cm] (mxp) at (16.5, 9.0) {M\\U\\X};
\node[cmb, minimum width=2.2cm] (pat) at (16.5, 11.0) {Patrón de\\calibración (fijo)};
\node[ext] (sw) at (18.9, 9.0) {DIP SW2};
\draw[bus] (and.east) -- node[lbl, above] {256} (mxp.west);
\draw[bus] (pat.south) -- (mxp.north);
\draw[sig] (sw.west) -- node[lbl, above] {sel} (mxp.east);

% ======================= (C) Transmisor MAX7219 =======================
\node[reg, minimum width=2.7cm] (buf) at (16.5, 5.8) {Registro de imagen\\256 bits};
\draw[bus] (mxp.south) -- node[lbl, right] {bytes[255:0]} (buf.north);
\node[cmb, minimum width=2.7cm] (rom) at (16.5, 4.2) {ROM de configuración\\(5 palabras fijas)};

\node[mux, minimum height=2.4cm] (mxw) at (12.0, 5.0) {M\\U\\X};
\draw[bus] (buf.west) -- node[lbl, above] {fila $d$: 1 byte\\por chip} (buf.west -| mxw.east);
\draw[bus] (rom.west) -- node[lbl, below] {palabra fija} (rom.west -| mxw.east);

\node[reg, minimum width=2.6cm] (sr) at (8.3, 5.0) {Registro de\\desplazamiento\\64 bits};
\draw[bus] (mxw.west) -- node[lbl, above] {trama} node[lbl, below] {64 bits} (sr.east);

\node[reg] (cidx) at (12.0, 2.4) {Contador idx\\(0 … 12)};
\draw[sig] (cidx.north) -- node[lbl, right] {sel} (mxw.south);

\node[fsm] (fsm) at (4.0, 2.4) {FSM del transmisor\\CARGA, BAJO, ALTO,\\CIERRE, LATCH, SEPARA};
\node[reg, minimum width=2.0cm] (cbit) at (0.6, 5.0) {Contador\\de bits\\(64 … 1)};
\node[reg, minimum width=2.0cm] (cdiv) at (0.6, 0.6) {Contador de\\medio periodo};
\draw[sig] (cbit.south) |- node[lbl, pos=0.25, left] {U} ([yshift=3mm]fsm.west);
\draw[sig] (cdiv.north) |- node[lbl, pos=0.25, left] {T} ([yshift=-3mm]fsm.west);
\draw[sig] (fsm.north) |- node[lbl, pos=0.75, above] {cargar, desplazar} (sr.west);
\draw[sig] (fsm.east) -- node[lbl, above, pos=0.7] {siguiente} (cidx.west);

% Salidas hacia el hardware
\node[ext, minimum width=6.2cm] (out) at (6.3, -0.9) {SN74AHCT125 $\rightarrow$ módulo MAX7219 ×4};
\draw[sig] (sr.south) -- node[lbl, right, pos=0.12] {DIN = bit 63} (sr.south |- out.north);
\draw[sig] ([xshift=-7mm]fsm.south) -- node[lbl, left] {CLK\_LED\\= ALTO} ([xshift=-7mm]fsm.south |- out.north);
\draw[sig] ([xshift=7mm]fsm.south) -- node[lbl, right] {LOAD\\= LATCH} ([xshift=7mm]fsm.south |- out.north);

% ======================= Cajas y títulos =======================
\begin{scope}[on background layer]
  \node[caja, fit=(clk)(div)(pos), label={[tit, anchor=south west]north west:Posición del punto}] {};
  \node[caja, fit=(dy)(db)(and)(mxp)(pat), label={[tit, anchor=south west]north west:Generador de imagen (combinacional)}] {};
  \node[caja, fit=(cbit)(cdiv)(fsm)(sr)(buf)(rom)(cidx), label={[tit, anchor=south west]north west:Transmisor serial MAX7219 (secuencial)}] {};
\end{scope}
\end{tikzpicture}
